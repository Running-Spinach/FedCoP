# --- FedCoP 完整方法 (benchmark 超参) ---
python code/exps/federated_main.py --alg fedcop \
    --dataset chestxray14 --num_classes 14 \

    --ent_lambda 1e-3 --proto_momentum 0.9 --temperature 1.0 \
    --ld_warmup 10 --co_warmup 5 --fuse_alpha 0.5 \
    --seed 1234

# R̂ = I (关共现结构)

    --co_lambda 0.1 --cov_shrinkage 0.1 --co_beta 1.0 --co_mf_steps 2 \
    --ent_lambda 1e-3 --proto_momentum 0.9 --temperature 1.0 --ld_warmup 10 \
    --no_cooccurrence

    --ent_lambda 1e-3 --proto_momentum 0.9 --temperature 1.0 --ld_warmup 10 \
    --local_cooc_only

# 关训练侧 L_co

# 七、完整 CLI 参数速查 (来自 code/lib/options.py)
# ═══════════════════════════════════════════════════════════════════════════════

# --- 联邦学习基础参数 ---
# --rounds 100            全局通信轮数
# --num_users 20          客户端数量 K
# --frac 0.25             每轮参与客户端比例
# --train_ep 1            本地训练 epoch 数
# --local_bs 4            本地 batch size
# --lr 0.01               学习率
# --momentum 0.5           SGD 动量

# --- 模型与算法 ---
# --alg fedcop            fedavg | fedprox | fedproto | fedgmkd | fedseproto | fedcop
# --model resnet50        resnet50 | cnn
# --norm batch_norm       batch_norm | layer_norm | None
# --num_filters 32        仅 CNN
# --max_pool True          仅 CNN

# --- 数据集 ---
# --dataset chestxray14   chestxray14 | mured
# --num_classes 14        14 (ChestX-ray14) | 20 (MuReD)
# --data_dir ./data/      数据集根目录
# --image_size 224        输入图像尺寸

# --- 硬件 / 训练 ---
# --gpu 0                 GPU ID (默认 CPU)
# --optimizer sgd         sgd | adam
# --seed 1234             随机种子
# --iid 0                 0=Non-IID | 1=IID
# --unequal 0             0=等量 | 1=不等量
# --stopping_rounds 10    早停容忍轮数
# --test_ep 10            测试间隔轮数
# --verbose 1             0=静默 | 1=详细

# --- Non-IID 划分 ---
# --ways 3                每客户端类别数 (越小越 Non-IID)
# --shots 100             每类样本数 (每个客户端)
# --train_shots_max 200   每类最大训练样本数
# --test_shots 15         每类测试样本数
# --stdev 2               类别数/样本数的标准差

# --- 原型学习 (FedProto / FedCoP 共用) ---
# --ld 1.0                原型损失权重 λ
# --ld_warmup 50          原型损失 warmup 轮数
# --proto_momentum 0.9    EMA 动量 (原型 + R̂)
# --ft_round 10           微调轮数 (实验性)

# --- 分布原型 ---
# --use_distributional    启用高斯原型 (FedCoP 始终启用)
# --dist_type kl          kl | wasserstein | mse
# --proto_dim None        None=自动 | 正整数

# --- FedCoP 共现结构 (核心创新) ---
# --co_lambda 0.1          L_co 权重
# --co_warmup 10           L_co warmup 轮数
# --cov_shrinkage 0.1      R̂ 收缩系数 η
# --co_rank 0              R̂ 低秩近似 (0=全秩)
# --co_beta 1.0            mean-field 耦合强度 β
# --co_mf_steps 2          mean-field 迭代步数
# --ent_lambda 0.001       熵正则权重
# --fuse_alpha 0.5         分类器/原型融合权重 α
# --temperature 1.0        推理温度 T

# --- FedCoP 消融开关 ---
# --no_cooccurrence        R̂=I, 关共现结构
# --local_cooc_only        只用本地共现, 不联邦聚合
# --no_lco                 关训练侧 L_co, 保留推理侧 R̂

# --- 基线专属参数 ---
# --fedprox_mu 0.01        FedProx 近端正则系数
# --gmm_components 3       FedGMKD GMM 分量数 K
# --mi_lambda 0.05         FedSeProto 互信息最小化权重


# ═══════════════════════════════════════════════════════════════════════════════
# 八、算法列表总览
# ═══════════════════════════════════════════════════════════════════════════════

# 基线方法 (5个):
#   fedavg       — 权重平均 (AISTATS 2017)
#   fedprox      — 权重平均 + 近端正则 (MLSys 2020)
#   fedproto     — 点原型 + 最近邻 (AAAI 2022)
#   fedgmkd      — GMM 原型 (NeurIPS 2024)
#   fedseproto   — 语义域解耦 (ECAI 2024)

# 提出方法:
#   fedcop       — 共现感知分布原型 (Proposed)
#     fedcop_nocoo   — 消融: R̂=I (关共现结构)
#     fedcop_local   — 消融: 非联邦 R (只用本地)
#     fedcop_nolco   — 消融: 关训练侧 L_co

# 消融变体 (A0-A9):
#   A0: 完整方法
#   A1: no L_co (训练侧共现对齐)
#   A2: no structure decode (R=I)
#   A3: point proto (MSE, 无熵)
#   A4: no L_ent
#   A5: decode=纯分类器 (α=1)
#   A6: decode=纯原型 (α=0)
#   A7: local cooc (非联邦 R)
#   A8: W2 距离 (替代 KL)
#   A9: temperature=10 (诊断)
